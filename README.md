🚀 Использование

chmod +x dsh.sh

./dsh.sh           # поднять, дождаться токена и открыть браузер
./dsh.sh up        # поднять, дождаться токена и открыть браузер
./dsh.sh rebuild   # пересобрать образ и открыть браузер
./dsh.sh url       # просто вывести URL (например, для копирования)
./dsh.sh open      # открыть текущую сессию в браузере
./dsh.sh logs      # смотреть логи в realtime
./dsh.sh restart   # перезапустить и открыть
./dsh.sh down      # остановить
